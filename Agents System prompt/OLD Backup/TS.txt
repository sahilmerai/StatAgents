code = """
import pandas as pd

df = pd.read_csv('/data/Tractor_Sales.csv')
df['Month-Year'] = pd.to_datetime(df['Month-Year'])
df = df.sort_values('Month-Year').set_index('Month-Year')

ts_agg = df[target_col].resample(AGGREGATION_LEVEL).sum()

print("DURBIN-WATSON TEST FOR AUTOCORRELATION")
print("="*50)
print("H0: No autocorrelation (residuals are independent)")
print("H1: Autocorrelation exists (residuals are correlated)")
print()
print(f"Durbin-Watson statistic: {dw:.4f}")
print()
print("Interpretation:")
print("- DW ≈ 2: No autocorrelation (H0 accepted)")
print("- DW < 2: Positive autocorrelation (H1 accepted)")
print("- DW > 2: Negative autocorrelation (H1 accepted)")
print()
if abs(dw - 2) < 0.5:
    print(f"Result: DW={dw:.4f} ≈ 2 → No significant autocorrelation")
elif dw < 2:
    print(f"Result: DW={dw:.4f} < 2 → Positive autocorrelation detected")
else:
    print(f"Result: DW={dw:.4f} > 2 → Negative autocorrelation detected")

**STEP 9: Breusch-Godfrey Test**

code = """
import pandas as pd
import statsmodels.api as sm
from statsmodels.stats.diagnostic import acorr_breusch_godfrey

df = pd.read_csv('/data/Tractor_Sales.csv')
df['Month-Year'] = pd.to_datetime(df['Month-Year'])
df = df.sort_values('Month-Year').set_index('Month-Year')

target_col = 'Number of Tractor Sold'
AGGREGATION_LEVEL = 'M'
ts_agg = df[target_col].resample(AGGREGATION_LEVEL).sum()

# Fit simple OLS for residuals
X = sm.add_constant(range(len(ts_agg)))
y = ts_agg.values
model = sm.OLS(y, X).fit()

# Breusch-Godfrey test
bg_test = acorr_breusch_godfrey(model, nlags=5)

print("BREUSCH-GODFREY TEST FOR AUTOCORRELATION")
print("="*50)
print("H0: No autocorrelation up to lag order p")
print("H1: Autocorrelation exists up to lag order p")
print()
print(f"LM Statistic: {bg_test[0]:.4f}")
print(f"p-value: {bg_test[1]:.4f}")
print()
print("Interpretation:")
if bg_test[1] > 0.05:
    print(f"p={bg_test[1]:.4f} > 0.05 → Fail to reject H0")
    print("Result: No significant autocorrelation detected")
else:
    print(f"p={bg_test[1]:.4f} ≤ 0.05 → Reject H0")
    print("Result: Significant autocorrelation detected")
print("="*50)

**STEP 11: ADF Test for Stationarity**

code = """
import pandas as pd
from statsmodels.tsa.stattools import adfuller

df = pd.read_csv('/data/Tractor_Sales.csv')

AGGREGATION_LEVEL = 'M'
ts_agg = df[target_col].resample(AGGREGATION_LEVEL).sum()

# ADF Test
adf_result = adfuller(ts_agg, autolag='AIC')

print("AUGMENTED DICKEY-FULLER TEST FOR STATIONARITY")
print("="*50)
print("H0: Series has a unit root (non-stationary)")
print("H1: Series has no unit root (stationary)")
print()
print(f"ADF Statistic: {adf_result[0]:.4f}")
print(f"p-value: {adf_result[1]:.4f}")
print(f"Lags used: {adf_result[2]}")
print(f"Observations: {adf_result[3]}")
print()
print("Critical Values:")
for key, value in adf_result[4].items():
    print(f"  {key}: {value:.4f}")
print()
print("Interpretation:")
if adf_result[1] <= 0.05:
    print(f"p={adf_result[1]:.4f} ≤ 0.05 → Reject H0")
    print("Result: Series is STATIONARY")
