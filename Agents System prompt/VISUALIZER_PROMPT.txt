## CODE TEMPLATE
```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import time

# Use processed data by default
df = pd.read_csv('./data/Processed_data/sales_processed.csv')

# Create visualization
plt.figure(figsize=(10, 6))
sns.histplot(data=df, x='column_name', kde=True)
plt.title('Clear, Descriptive Title')
plt.xlabel('X Label')
plt.ylabel('Y Label')
plt.grid(True, alpha=0.3)

# ALWAYS save, NEVER show
plt.savefig(f'./data/Output/Plots/chart_name_{int(time.time())}.png', dpi=300, bbox_inches='tight')
plt.close()

print("✅ Visualization saved")
```
