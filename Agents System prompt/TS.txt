### STEP 6: Seasonal Decomposition
```python
from statsmodels.tsa.seasonal import seasonal_decompose

# Determine period based on aggregation
period_map = {'D': 7, 'W': 52, 'M': 12, 'Q': 4}
period = period_map.get(aggregation, 12)

print(f"\nRunning seasonal decomposition with period={period}")

decomp = seasonal_decompose(ts, model='additive', period=period)

fig = decomp.plot()
fig.set_size_inches(14, 10)
plt.tight_layout()
plt.savefig('./data/Output/Plots/ts_decomposition.png')
plt.close()

print("\nDecomposition complete")
print("Components: Trend, Seasonal, Residual")
print(f"Seasonal strength: {decomp.seasonal.std() / ts.std():.2%}")
```

**Describe the decomposition results, then proceed to Step 7.**

### STEP 7: Stationarity Test (ADF)
```python
from statsmodels.tsa.stattools import adfuller

adf_result = adfuller(ts, autolag='AIC')

print("\n" + "="*60)
print("AUGMENTED DICKEY-FULLER TEST")
print("="*60)
print(f"ADF Statistic: {adf_result[0]:.4f}")
print(f"p-value: {adf_result[1]:.4f}")
print(f"Critical values:")
for key, value in adf_result[4].items():
    print(f"  {key}: {value:.4f}")

if adf_result[1] <= 0.05:
    print("\n✓ Series is STATIONARY (p ≤ 0.05)")
    print("Can proceed with ARIMA modeling")
else:
    print("\n⚠ Series is NON-STATIONARY (p > 0.05)")
    print("Will need differencing (d≥1 in ARIMA)")
```

### STEP 8: ACF/PACF Plots

```python
from statsmodels.graphics.tsaplots import plot_acf, plot_pacf
```

### STEP 11: Forecast & Evaluate
```python
# Forecast on test set
forecast = model.predict(n_periods=len(test))
forecast_series = pd.Series(forecast, index=test.index)

# Calculate accuracy metrics
from sklearn.metrics import mean_absolute_error, mean_squared_error

mae = mean_absolute_error(test, forecast)
rmse = np.sqrt(mean_squared_error(test, forecast))
mape = np.mean(np.abs((test - forecast) / test)) * 100

print("\n" + "="*60)
print("FORECAST ACCURACY")
print("="*60)
print(f"MAE:  {mae:.2f}")
print(f"RMSE: {rmse:.2f}")
print(f"MAPE: {mape:.2f}%")

# Interpretation
if mape < 10:
    print("\n✓ Excellent accuracy (MAPE < 10%)")
elif mape < 20:
    print("\n✓ Good accuracy (MAPE < 20%)")
else:
    print("\n⚠ Moderate accuracy (MAPE ≥ 20%) - consider model improvements")

# Plot forecast vs actual
plt.figure(figsize=(14, 6))
plt.plot(train.index, train, label='Training Data', linewidth=1.5)
plt.plot(test.index, test, label='Actual Test Data', linewidth=1.5, color='green')
plt.plot(forecast_series.index, forecast_series, label='Forecast', linewidth=1.5, 
         linestyle='--', color='red')
plt.title('Time Series Forecast vs Actual')
```
